if JAX_AVAILABLE:
    from functools import partial

    @partial(jit, static_argnums=(5, 6, 7, 8, 9))
    def _adam_update_matrix_jax(W, dW, M, R, mask, lrate, beta1, beta2, eps, epoch_num):
        """
        JIT-compiled Adam update for weight matrix on GPU.

        Args:
            W: Current weights (JAX array on GPU)
            dW: Gradients (JAX array on GPU)
            M, R: Adam momentum states (JAX arrays on GPU)
            mask: Update mask (JAX array on GPU or None)
            lrate, beta1, beta2, eps, epoch_num: Scalars (static)

        Returns:
            Updated W, M, R (all on GPU)
        """
        # Update momentum
        M = beta1 * M + (1.0 - beta1) * dW

        # Update RMSprop
        R = beta2 * R + (1.0 - beta2) * (dW ** 2)

        # Bias correction
        m_hat = M / (1.0 - beta1 ** epoch_num)
        r_hat = R / (1.0 - beta2 ** epoch_num)

        # Compute update
        update = lrate * m_hat / (jnp.sqrt(r_hat) + eps)

        # Apply mask if provided
        if mask is not None:
            update = update * mask

        # Update weights
        W_new = W + update

        return W_new, M, R

    @partial(jit, static_argnums=(3,))
    def _sgd_update_matrix_jax(W, dW, mask, lrate):
        """
        JIT-compiled SGD update for weight matrix on GPU.
        """
        if mask is not None:
            update = lrate * dW * mask
        else:
            update = lrate * dW
        return W + update

    @jit
    def _compute_weight_decay_jax(W, W_ref, factor):
        """
        JIT-compiled weight decay computation on GPU.
        """
        return -factor * (W - W_ref)
